# Skills App — Plan 3: Gitarre, Audio und Song-Import

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ein vollständiger Lernbereich E-Gitarre — 13 aufeinander aufbauende Lektionen mit Handanweisungen, Griffbrett- und Akkord-Widgets, Stimmgerät und Metronom über das Mikrofon, eine Songbibliothek mit Übungsstücken sowie Song-Import per Text-Einfügen und per URL.

**Architecture:** Alles Musikalische, das reine Rechnung ist — Notennamen, Intervalle, Akkordformeln, Transponieren, Tab- und Akkord-Parser, Tonhöhenerkennung — liegt in `packages/core` und ist ohne Browser testbar. Alles, was Geräte braucht — Mikrofon, Audioausgabe — liegt in einem eigenen Web-Modul `packages/web/src/audio`, das die reinen Funktionen aus `core` aufruft. Der Lernbereich unter `skills/guitar/` liefert Inhalte und vier Widgets. Der Kern bekommt genau eine Ergänzung: die Song-Routen im Server.

**Tech Stack:** Zusätzlich zu Plan 1: Web Audio API (`AudioContext`, `AnalyserNode`, `getUserMedia`) ohne Bibliothek, `undici` für den serverseitigen URL-Abruf, `cheerio` für die HTML-Extraktion.

```ts
export const NOTES = ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"] as const;
const FLAT_NOTES = ["C", "Db", "D", "Eb", "E", "F", "Gb", "G", "Ab", "A", "Bb", "B"] as const;

const INTERVALS = [
  "Prime",
  "kleine Sekunde",
  "große Sekunde",
  "kleine Terz",
  "große Terz",
  "Quarte",
  "Tritonus",
  "Quinte",
  "kleine Sexte",
  "große Sexte",
  "kleine Septime",
  "große Septime",
  "Oktave",
] as const;

export interface Tuning {
  name: string;
  strings: string[];
}

export const TUNINGS: Tuning[] = [
  { name: "Standard", strings: ["E2", "A2", "D3", "G3", "B3", "E4"] },
  { name: "Drop D", strings: ["D2", "A2", "D3", "G3", "B3", "E4"] },
  { name: "Half Step Down", strings: ["D#2", "G#2", "C#3", "F#3", "A#3", "D#4"] },
  { name: "Open G", strings: ["D2", "G2", "D3", "G3", "B3", "D4"] },
];

export function noteToSemitone(note: string): number {
  const sharp = NOTES.indexOf(note as (typeof NOTES)[number]);
  if (sharp >= 0) return sharp;
  const flat = FLAT_NOTES.indexOf(note as (typeof FLAT_NOTES)[number]);
  if (flat >= 0) return flat;
  throw new Error(`Unbekannter Ton: ${note}`);
}

export function semitoneToNote(semitone: number, preferFlats = false): string {
  const index = ((semitone % 12) + 12) % 12;
  return preferFlats ? FLAT_NOTES[index]! : NOTES[index]!;
}

export function transposeNote(note: string, steps: number, preferFlats = false): string {
  return semitoneToNote(noteToSemitone(note) + steps, preferFlats);
}

function splitOctave(noteWithOctave: string): { note: string; octave: number } {
  const match = /^([A-G][#b]?)(-?\d+)$/.exec(noteWithOctave);
  if (!match) throw new Error(`Ton ohne Oktave: ${noteWithOctave}`);
  return { note: match[1]!, octave: Number(match[2]) };
}

/** Absolute pitch index; C0 is 0, so A4 lands on 57. */
function absoluteIndex(noteWithOctave: string): number {
  const { note, octave } = splitOctave(noteWithOctave);
  return octave * 12 + noteToSemitone(note);
}

function fromAbsoluteIndex(index: number, preferFlats = false): string {
  const octave = Math.floor(index / 12);
  return `${semitoneToNote(index % 12, preferFlats)}${octave}`;
}

export function noteAt(tuning: Tuning, stringIndex: number, fret: number): string {
  const open = tuning.strings[stringIndex];
  if (!open) throw new Error(`Saite ${stringIndex} gibt es in dieser Stimmung nicht.`);
  return fromAbsoluteIndex(absoluteIndex(open) + fret);
}

const A4_INDEX = absoluteIndex("A4");

export function frequencyOf(noteWithOctave: string): number {
  return 440 * Math.pow(2, (absoluteIndex(noteWithOctave) - A4_INDEX) / 12);
}

export function nearestNote(frequency: number): { note: string; cents: number } {
  const exact = A4_INDEX + 12 * Math.log2(frequency / 440);
  const rounded = Math.round(exact);
  return {
    note: fromAbsoluteIndex(rounded),
    cents: Math.round((exact - rounded) * 100),
  };
}

export function intervalName(semitones: number): string {
  const name = INTERVALS[semitones];
  if (name) return name;
  return `${semitones} Halbtöne`;
}
```

```ts
import { noteToSemitone, semitoneToNote, transposeNote } from "./notes.js";

export interface ParsedChord {
  root: string;
  quality: string;
  bass: string | null;
}

export interface ChordShape {
  name: string;
  frets: (number | null)[];
  fingers: (number | null)[];
  baseFret: number;
}

const CHORD_PATTERN = /^([A-G][#b]?)([^/\s]*)(?:\/([A-G][#b]?))?$/;

const KNOWN_QUALITIES = new Set([
  "", "m", "maj7", "m7", "7", "6", "m6", "9", "m9", "dim", "aug",
  "sus2", "sus4", "add9", "5", "7sus4", "m7b5",
]);

export function parseChord(symbol: string): ParsedChord | null {
  const match = CHORD_PATTERN.exec(symbol.trim());
  if (!match) return null;
  const quality = match[2] ?? "";
  if (!KNOWN_QUALITIES.has(quality)) return null;
  return { root: match[1]!, quality, bass: match[3] ?? null };
}

export function transposeChord(symbol: string, steps: number, preferFlats = false): string {
  const parsed = parseChord(symbol);
  if (!parsed) return symbol;
  const root = transposeNote(parsed.root, steps, preferFlats);
  const bass = parsed.bass === null ? null : transposeNote(parsed.bass, steps, preferFlats);
  return `${root}${parsed.quality}${bass ? `/${bass}` : ""}`;
}

const FORMULAS: Record<string, number[]> = {
  "": [0, 4, 7],
  m: [0, 3, 7],
  maj7: [0, 4, 7, 11],
  m7: [0, 3, 7, 10],
  "7": [0, 4, 7, 10],
  "6": [0, 4, 7, 9],
  m6: [0, 3, 7, 9],
  "9": [0, 4, 7, 10, 14],
  m9: [0, 3, 7, 10, 14],
  dim: [0, 3, 6],
  aug: [0, 4, 8],
  sus2: [0, 2, 7],
  sus4: [0, 5, 7],
  add9: [0, 4, 7, 14],
  "5": [0, 7],
  "7sus4": [0, 5, 7, 10],
  m7b5: [0, 3, 6, 10],
};

export function chordNotes(symbol: string): string[] {
  const parsed = parseChord(symbol);
  if (!parsed) return [];
  const formula = FORMULAS[parsed.quality] ?? FORMULAS[""]!;
  const root = noteToSemitone(parsed.root);
  return formula.map((step) => semitoneToNote(root + step));
}

const MAJOR_SCALE = [0, 2, 4, 5, 7, 9, 11];
const DEGREES = ["I", "ii", "iii", "IV", "V", "vi", "vii°"];
const QUALITIES = ["", "m", "m", "", "", "m", "dim"];

export function keyChords(root: string): { grad: string; akkord: string }[] {

  return MAJOR_SCALE.map((step, i) => ({
    grad: DEGREES[i]!,
    akkord: `${semitoneToNote(base + step)}${QUALITIES[i]}`,

}
```

```ts
import { z } from "zod";
import { parseChord, transposeChord } from "./chords.js";

export type SongFormat = "chordpro" | "chordsheet" | "tab";

export const songLineSchema = z.object({
  lyrics: z.string(),
  chords: z.array(z.object({ position: z.number().int().nonnegative(), symbol: z.string() })),
});

export const tabLineSchema = z.object({ strings: z.array(z.string()).length(6) });

export const songSectionSchema = z.object({
  name: z.string(),
  lines: z.array(songLineSchema),
  tab: z.array(tabLineSchema),
});

export const songSchema = z.object({
  title: z.string().min(1),
  artist: z.string().nullable(),
  capo: z.number().int().nonnegative(),
  sections: z.array(songSectionSchema),
  chordsUsed: z.array(z.string()),
});

export type SongLine = z.infer<typeof songLineSchema>;
export type TabLine = z.infer<typeof tabLineSchema>;
export type SongSection = z.infer<typeof songSectionSchema>;
export type Song = z.infer<typeof songSchema>;

export function collectChords(sections: SongSection[]): string[] {
  const seen = new Set<string>();
  for (const section of sections) {
    for (const line of section.lines) {
      for (const chord of line.chords) seen.add(chord.symbol);
    }
  }
  return [...seen].sort();
}

export function transposeSong(song: Song, steps: number, preferFlats = false): Song {
  const sections = song.sections.map((section) => ({
    ...section,
    lines: section.lines.map((line) => ({
      ...line,
      chords: line.chords.map((chord) => ({
        ...chord,
        symbol: transposeChord(chord.symbol, steps, preferFlats),
      })),
    })),
  }));
  return { ...song, sections, chordsUsed: collectChords(sections) };
}

export function isChordToken(token: string): boolean {
  return parseChord(token) !== null;
}
```

```ts
import type { SongFormat } from "./song.js";

const TAB_LINE = /^\s*[eEBGDAa][|:]/;

export function detectFormat(text: string): SongFormat {
  const lines = text.split(/\r?\n/);

  if (lines.filter((line) => TAB_LINE.test(line)).length >= 4) return "tab";
  if (lines.some((line) => /^\s*\{[a-z_]+\s*:?/i.test(line))) return "chordpro";
  if (lines.some((line) => /\[[A-G][#b]?[^\]]*\]\S/.test(line))) return "chordpro";

  return "chordsheet";
}
```

```ts
import { collectChords, type Song, type SongSection } from "./song.js";

const DIRECTIVE = /^\s*\{\s*([a-z_]+)\s*(?::\s*(.*?))?\s*\}\s*$/i;
const INLINE_CHORD = /\[([^\]]+)\]/g;

const SECTION_NAMES: Record<string, string> = {
  verse: "Strophe",
  chorus: "Refrain",
  bridge: "Bridge",
  intro: "Intro",
  outro: "Outro",
};

export function parseChordPro(text: string): Song {
  let title = "Ohne Titel";
  let artist: string | null = null;
  let capo = 0;

  const sections: SongSection[] = [];
  let current: SongSection = { name: "Song", lines: [], tab: [] };

  const push = () => {
    if (current.lines.length > 0 || current.tab.length > 0) sections.push(current);
  };

  for (const raw of text.split(/\r?\n/)) {
    const directive = DIRECTIVE.exec(raw);
    if (directive) {
      const key = directive[1]!.toLowerCase();
      const value = directive[2]?.trim() ?? "";

      if (key === "title" || key === "t") title = value;
      else if (key === "artist" || key === "subtitle" || key === "st") artist = value;
      else if (key === "capo") capo = Number(value) || 0;
      else if (key.startsWith("start_of_")) {
        push();
        const suffix = key.slice("start_of_".length);
        current = { name: SECTION_NAMES[suffix] ?? suffix, lines: [], tab: [] };
      } else if (key.startsWith("end_of_")) {
        push();
        current = { name: "Song", lines: [], tab: [] };
      }
      continue;
    }

    if (raw.trim() === "") continue;

    const chords: { position: number; symbol: string }[] = [];
    let lyrics = "";
    let cursor = 0;
    INLINE_CHORD.lastIndex = 0;

    let match: RegExpExecArray | null;
    while ((match = INLINE_CHORD.exec(raw)) !== null) {
      lyrics += raw.slice(cursor, match.index);
      chords.push({ position: lyrics.length, symbol: match[1]!.trim() });
      cursor = match.index + match[0].length;
    }
    lyrics += raw.slice(cursor);

    current.lines.push({ lyrics: lyrics.trim(), chords });
  }

  push();
  if (sections.length === 0) sections.push({ name: "Song", lines: [], tab: [] });

  return { title, artist, capo, sections, chordsUsed: collectChords(sections) };
}
```

`packages/core/src/music/parseChordSheet.ts`:

```ts
import { collectChords, isChordToken, type Song, type SongLine, type SongSection } from "./song.js";

const SECTION_HEADER = /^\s*\[([^\]]+)\]\s*$/;

/** A line counts as a chord line when every token on it parses as a chord. */
function isChordLine(line: string): boolean {
  const tokens = line.trim().split(/\s+/).filter(Boolean);
  if (tokens.length === 0) return false;
  return tokens.every(isChordToken);
}

function chordsWithPositions(line: string): { position: number; symbol: string }[] {
  const result: { position: number; symbol: string }[] = [];
  const pattern = /\S+/g;
  let match: RegExpExecArray | null;
  while ((match = pattern.exec(line)) !== null) {
    result.push({ position: match.index, symbol: match[0] });
  }
  return result;
}

export function parseChordSheet(text: string): Song {
  const lines = text.split(/\r?\n/);

  let index = 0;
  while (index < lines.length && lines[index]!.trim() === "") index += 1;

  const title = lines[index]?.trim() || "Ohne Titel";
  index += 1;

  let artist: string | null = null;
  const candidate = lines[index]?.trim() ?? "";
  if (candidate !== "" && !SECTION_HEADER.test(candidate) && !isChordLine(candidate)) {
    artist = candidate;
    index += 1;
  }

  const sections: SongSection[] = [];
  let current: SongSection = { name: "Song", lines: [], tab: [] };
  const push = () => {
    if (current.lines.length > 0) sections.push(current);
  };

  for (; index < lines.length; index += 1) {
    const raw = lines[index]!;

    const header = SECTION_HEADER.exec(raw);
    if (header) {
      push();
      current = { name: header[1]!.trim(), lines: [], tab: [] };
      continue;
    }

    if (raw.trim() === "") continue;

    if (isChordLine(raw)) {
      const chords = chordsWithPositions(raw);
      const next = lines[index + 1];
      if (next !== undefined && next.trim() !== "" && !isChordLine(next) && !SECTION_HEADER.test(next)) {
        current.lines.push({ lyrics: next.trimEnd(), chords });
        index += 1;
      } else {
        current.lines.push({ lyrics: "", chords });
      }
      continue;
    }

    const line: SongLine = { lyrics: raw.trimEnd(), chords: [] };
    current.lines.push(line);
  }

  push();
  if (sections.length === 0) sections.push({ name: "Song", lines: [], tab: [] });

  return { title, artist, capo: 0, sections, chordsUsed: collectChords(sections) };
}
```

```ts
import { collectChords, type Song, type SongSection, type TabLine } from "./song.js";

const TAB_LINE = /^\s*[eEBGDAa][|:]/;

export function parseTab(text: string): Song {
  const lines = text.split(/\r?\n/);

  let index = 0;
  while (index < lines.length && lines[index]!.trim() === "") index += 1;

  const title = lines[index]?.trim() || "Ohne Titel";
  index += 1;

  let artist: string | null = null;
  const candidate = lines[index]?.trim() ?? "";
  if (candidate !== "" && !TAB_LINE.test(candidate)) {
    artist = candidate;
    index += 1;
  }

  const tab: TabLine[] = [];
  let buffer: string[] = [];

  for (; index < lines.length; index += 1) {
    const raw = lines[index]!;
    if (TAB_LINE.test(raw)) {
      buffer.push(raw.trimEnd());
      if (buffer.length === 6) {
        tab.push({ strings: [...buffer] });
        buffer = [];
      }
      continue;
    }
    if (raw.trim() === "" && buffer.length > 0) {
      while (buffer.length < 6) buffer.push("");
      tab.push({ strings: [...buffer] });
      buffer = [];
    }
  }

  if (buffer.length > 0) {
    while (buffer.length < 6) buffer.push("");
    tab.push({ strings: buffer });
  }

  const sections: SongSection[] = [{ name: "Riff", lines: [], tab }];
  return { title, artist, capo: 0, sections, chordsUsed: collectChords(sections) };
}
```

```ts
import { frequencyOf, nearestNote, type Tuning } from "../music/notes.js";

export interface PitchOptions {
  threshold?: number;
  minFrequency?: number;
  maxFrequency?: number;
  minRms?: number;
}

export interface TuningReading {
  frequency: number;
  note: string;
  cents: number;
  targetString: number | null;
}

function rms(buffer: Float32Array): number {
  let sum = 0;
  for (let i = 0; i < buffer.length; i += 1) sum += buffer[i]! * buffer[i]!;
  return Math.sqrt(sum / buffer.length);
}

/**
 * YIN pitch detection. Steps 1-4 of the paper: difference function,
 * cumulative mean normalisation, absolute threshold, parabolic interpolation.
 */
export function detectPitch(
  buffer: Float32Array,
  sampleRate: number,
  options: PitchOptions = {},
): number | null {
  const threshold = options.threshold ?? 0.15;
  const minFrequency = options.minFrequency ?? 60;
  const maxFrequency = options.maxFrequency ?? 1400;
  const minRms = options.minRms ?? 0.01;

  if (rms(buffer) < minRms) return null;

  const minLag = Math.floor(sampleRate / maxFrequency);
  const maxLag = Math.min(Math.floor(sampleRate / minFrequency), Math.floor(buffer.length / 2));
  if (maxLag <= minLag) return null;

  const difference = new Float32Array(maxLag + 1);
  for (let lag = minLag; lag <= maxLag; lag += 1) {
    let sum = 0;
    for (let i = 0; i < maxLag; i += 1) {
      const delta = buffer[i]! - buffer[i + lag]!;
      sum += delta * delta;
    }
    difference[lag] = sum;
  }

  const normalised = new Float32Array(maxLag + 1);
  normalised[0] = 1;
  let runningSum = 0;
  for (let lag = 1; lag <= maxLag; lag += 1) {
    runningSum += difference[lag]!;
    normalised[lag] = runningSum === 0 ? 1 : (difference[lag]! * lag) / runningSum;
  }

  let bestLag = -1;
  for (let lag = minLag; lag <= maxLag; lag += 1) {
    if (normalised[lag]! < threshold) {
      while (lag + 1 <= maxLag && normalised[lag + 1]! < normalised[lag]!) lag += 1;
      bestLag = lag;
      break;
    }
  }

  if (bestLag === -1) {
    let lowest = Number.POSITIVE_INFINITY;
    for (let lag = minLag; lag <= maxLag; lag += 1) {
      if (normalised[lag]! < lowest) {
        lowest = normalised[lag]!;
        bestLag = lag;
      }
    }
    if (lowest > 0.5) return null;
  }

  const previous = normalised[bestLag - 1] ?? normalised[bestLag]!;
  const next = normalised[bestLag + 1] ?? normalised[bestLag]!;
  const denominator = 2 * (2 * normalised[bestLag]! - previous - next);
  const refinedLag = denominator === 0 ? bestLag : bestLag + (next - previous) / denominator;

  const frequency = sampleRate / refinedLag;
  if (frequency < minFrequency || frequency > maxFrequency) return null;
  return frequency;
}

export function readTuning(
  buffer: Float32Array,
  sampleRate: number,
  tuning: Tuning,
  options: PitchOptions = {},
): TuningReading | null {
  const frequency = detectPitch(buffer, sampleRate, options);
  if (frequency === null) return null;

  const { note, cents } = nearestNote(frequency);

  let targetString: number | null = null;
  let smallest = Number.POSITIVE_INFINITY;
  tuning.strings.forEach((stringNote, index) => {
    const distance = Math.abs(1200 * Math.log2(frequency / frequencyOf(stringNote)));
    if (distance < smallest) {
      smallest = distance;
      targetString = index;
    }
  });

  // More than three semitones away means the player is not on that string at all.
  if (smallest > 300) targetString = null;

  return { frequency, note, cents, targetString };
}
```

```ts
import { useCallback, useEffect, useRef, useState } from "react";

export type MicState = "idle" | "requesting" | "running" | "denied" | "unsupported" | "error";

interface Options {
  onBuffer: (buffer: Float32Array, sampleRate: number) => void;
  fftSize?: number;
}

export function useMicrophone(options: Options) {
  const [state, setState] = useState<MicState>("idle");
  const [error, setError] = useState<string | null>(null);

  const contextRef = useRef<AudioContext | null>(null);
  const streamRef = useRef<MediaStream | null>(null);
  const frameRef = useRef<number | null>(null);
  const callbackRef = useRef(options.onBuffer);
  callbackRef.current = options.onBuffer;

  const stop = useCallback(() => {
    if (frameRef.current !== null) cancelAnimationFrame(frameRef.current);
    frameRef.current = null;
    streamRef.current?.getTracks().forEach((track) => track.stop());
    streamRef.current = null;
    void contextRef.current?.close();
    contextRef.current = null;
    setState("idle");
  }, []);

  const start = useCallback(async () => {
    if (typeof navigator === "undefined" || !navigator.mediaDevices?.getUserMedia) {
      setState("unsupported");
      setError("Dieser Browser bietet keinen Mikrofonzugriff an.");
      return;
    }

    setState("requesting");
    setError(null);

    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false },
      });
      streamRef.current = stream;

      const context = new AudioContext();
      contextRef.current = context;

      const source = context.createMediaStreamSource(stream);
      const analyser = context.createAnalyser();
      analyser.fftSize = options.fftSize ?? 4096;
      source.connect(analyser);

      const buffer = new Float32Array(analyser.fftSize);
      const tick = () => {
        analyser.getFloatTimeDomainData(buffer);
        callbackRef.current(buffer, context.sampleRate);
        frameRef.current = requestAnimationFrame(tick);
      };
      frameRef.current = requestAnimationFrame(tick);

      setState("running");
    } catch (e) {
      const name = e instanceof Error ? e.name : "";
      if (name === "NotAllowedError" || name === "SecurityError") {
        setState("denied");
        setError(
          "Der Zugriff auf das Mikrofon wurde abgelehnt. Erlaube ihn in den Browser-Einstellungen für diese Seite — der Rest der App funktioniert weiter.",
        );
      } else if (name === "NotFoundError") {
        setState("error");
        setError("Es wurde kein Mikrofon gefunden.");
      } else {
        setState("error");
        setError(e instanceof Error ? e.message : "Unbekannter Fehler beim Mikrofonzugriff.");
      }
    }
  }, [options.fftSize]);

  useEffect(() => stop, [stop]);

  return { state, error, start, stop };
}
```

`packages/web/src/audio/useMetronome.ts`:

```ts
import { useCallback, useEffect, useRef, useState } from "react";

const MIN_BPM = 20;
const MAX_BPM = 300;
const LOOKAHEAD_MS = 25;
const SCHEDULE_AHEAD_S = 0.1;

export function useMetronome() {
  const [bpm, setBpmRaw] = useState(80);
  const [beatsPerBar, setBeatsPerBar] = useState(4);
  const [running, setRunning] = useState(false);
  const [currentBeat, setCurrentBeat] = useState(0);

  const contextRef = useRef<AudioContext | null>(null);
  const nextNoteRef = useRef(0);
  const beatRef = useRef(0);
  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const tapsRef = useRef<number[]>([]);

  const bpmRef = useRef(bpm);
  bpmRef.current = bpm;
  const beatsRef = useRef(beatsPerBar);
  beatsRef.current = beatsPerBar;

  const setBpm = useCallback((value: number) => {
    setBpmRaw(Math.min(MAX_BPM, Math.max(MIN_BPM, Math.round(value))));
  }, []);

  const click = useCallback((time: number, accented: boolean) => {
    const context = contextRef.current;
    if (!context) return;
    const oscillator = context.createOscillator();
    const gain = context.createGain();
    oscillator.frequency.value = accented ? 1600 : 900;
    gain.gain.setValueAtTime(0.6, time);
    gain.gain.exponentialRampToValueAtTime(0.001, time + 0.05);
    oscillator.connect(gain);
    gain.connect(context.destination);
    oscillator.start(time);
    oscillator.stop(time + 0.06);
  }, []);

  const stop = useCallback(() => {
    if (timerRef.current !== null) clearInterval(timerRef.current);
    timerRef.current = null;
    void contextRef.current?.close();
    contextRef.current = null;
    beatRef.current = 0;
    setCurrentBeat(0);
    setRunning(false);
  }, []);

  const start = useCallback(() => {
    if (timerRef.current !== null) return;

    const context = new AudioContext();
    contextRef.current = context;
    void context.resume();
    nextNoteRef.current = context.currentTime + 0.05;
    beatRef.current = 0;

    // Scheduling ahead of time keeps the click steady; setInterval alone drifts.
    timerRef.current = setInterval(() => {
      const ctx = contextRef.current;
      if (!ctx) return;
      while (nextNoteRef.current < ctx.currentTime + SCHEDULE_AHEAD_S) {
        const beat = beatRef.current % beatsRef.current;
        click(nextNoteRef.current, beat === 0);
        setCurrentBeat(beat);
        nextNoteRef.current += 60 / bpmRef.current;
        beatRef.current += 1;
      }
    }, LOOKAHEAD_MS);

    setRunning(true);
  }, [click]);

  const tap = useCallback(() => {
    const now = Date.now();
    const taps = tapsRef.current.filter((t) => now - t < 3000);
    taps.push(now);
    tapsRef.current = taps;

    if (taps.length < 2) return;
    const gaps: number[] = [];
    for (let i = 1; i < taps.length; i += 1) gaps.push(taps[i]! - taps[i - 1]!);
    const average = gaps.reduce((sum, gap) => sum + gap, 0) / gaps.length;
    if (average > 0) setBpm(60_000 / average);
  }, [setBpm]);

  useEffect(() => stop, [stop]);

  return { bpm, setBpm, beatsPerBar, setBeatsPerBar, running, start, stop, tap, currentBeat };
}
```

```tsx
import { intervalName, noteAt, noteToSemitone, NOTES, TUNINGS } from "@skills/core";
import { useState } from "react";
import type { WidgetProps } from "../../../packages/web/src/skills/widgets.js";

type Mode = "noten" | "intervalle" | "tonleiter";

const FRET_COUNT = 15;
const MARKERS = [3, 5, 7, 9, 12, 15];
const MAJOR_SCALE = [0, 2, 4, 5, 7, 9, 11];
const SHORT_INTERVALS = ["1", "b2", "2", "b3", "3", "4", "b5", "5", "b6", "6", "b7", "7"];

export default function Fretboard(_: WidgetProps) {
  const [tuningName, setTuningName] = useState("Standard");
  const [mode, setMode] = useState<Mode>("noten");
  const [root, setRoot] = useState("E");

  const tuning = TUNINGS.find((t) => t.name === tuningName)!;
  const rootSemitone = noteToSemitone(root);

  function labelFor(stringIndex: number, fret: number): string {
    const withOctave = noteAt(tuning, stringIndex, fret);
    const note = withOctave.replace(/-?\d+$/, "");
    if (mode === "noten") return note;
    const distance = (noteToSemitone(note) - rootSemitone + 12) % 12;
    return SHORT_INTERVALS[distance]!;
  }

  function inScale(stringIndex: number, fret: number): boolean {
    if (mode !== "tonleiter") return true;
    const note = noteAt(tuning, stringIndex, fret).replace(/-?\d+$/, "");
    return MAJOR_SCALE.includes((noteToSemitone(note) - rootSemitone + 12) % 12);
  }

  return (
    <div className="space-y-5">
      <h1 className="text-2xl font-semibold">Griffbrett</h1>

      <div className="flex flex-wrap gap-4">
        <label className="text-sm">
          <span className="mr-2">Stimmung</span>
          <select
            value={tuningName}
            onChange={(e) => setTuningName(e.target.value)}
            className="rounded border border-slate-300 px-2 py-1 dark:border-slate-700 dark:bg-slate-900"
          >
            {TUNINGS.map((t) => (
              <option key={t.name} value={t.name}>
                {t.name}
              </option>
            ))}
          </select>
        </label>

        <label className="text-sm">
          <span className="mr-2">Anzeige</span>
          <select
            value={mode}
            onChange={(e) => setMode(e.target.value as Mode)}
            className="rounded border border-slate-300 px-2 py-1 dark:border-slate-700 dark:bg-slate-900"
          >
            <option value="noten">Notennamen</option>
            <option value="intervalle">Intervalle</option>
            <option value="tonleiter">Dur-Tonleiter</option>
          </select>
        </label>

        <label className="text-sm">
          <span className="mr-2">Grundton</span>
          <select
            value={root}
            onChange={(e) => setRoot(e.target.value)}
            className="rounded border border-slate-300 px-2 py-1 dark:border-slate-700 dark:bg-slate-900"
          >
            {NOTES.map((note) => (
              <option key={note} value={note}>
                {note}
              </option>
            ))}
          </select>
        </label>
      </div>

      <div className="overflow-x-auto">
        <table className="border-collapse text-center text-xs">
          <tbody>
            {[...tuning.strings].map((_, reversed) => {
              const stringIndex = tuning.strings.length - 1 - reversed;
              return (
                <tr key={stringIndex}>
                  {Array.from({ length: FRET_COUNT + 1 }, (_, fret) => {
                    const highlighted = inScale(stringIndex, fret);
                    return (
                      <td
                        key={fret}
                        data-testid={`fret-${stringIndex}-${fret}`}
                        data-in-scale={String(highlighted)}
                        className={`h-9 w-12 border border-slate-300 dark:border-slate-700 ${
                          fret === 0 ? "bg-slate-200 font-medium dark:bg-slate-800" : ""
                        } ${highlighted ? "" : "text-slate-300 dark:text-slate-700"}`}
                      >
                        {labelFor(stringIndex, fret)}
                      </td>
                    );
                  })}
                </tr>
              );
            })}
```

```ts
import { frequencyOf } from "@skills/core";

/** Plays a short sine at the pitch of the given note, for tuning by ear. */
export function playReferenceTone(noteWithOctave: string, seconds = 2): void {
  const context = new AudioContext();
  const oscillator = context.createOscillator();
  const gain = context.createGain();

  oscillator.type = "sine";
  oscillator.frequency.value = frequencyOf(noteWithOctave);
  gain.gain.setValueAtTime(0.0001, context.currentTime);
  gain.gain.exponentialRampToValueAtTime(0.3, context.currentTime + 0.05);
  gain.gain.exponentialRampToValueAtTime(0.0001, context.currentTime + seconds);

  oscillator.connect(gain);
  gain.connect(context.destination);
  oscillator.start();
  oscillator.stop(context.currentTime + seconds);
  oscillator.onended = () => void context.close();
}
```

```mdx
export const meta = { id: "01-haltung", title: "Instrument und Haltung", order: 1, requires: null };

<Ziel>Du sitzt und stehst so an der Gitarre, dass nichts verkrampft — die Grundlage für alles Weitere.</Ziel>

Wer sich hier Zeit lässt, spart sich Monate. Fast jede Blockade beim Üben — Finger, die nicht
weit genug greifen, ein Handgelenk, das nach zehn Minuten schmerzt — kommt aus der Haltung, nicht
aus mangelndem Talent.

**Im Sitzen:** Gitarre auf das rechte Bein (Rechtshänder), Korpus an den Körper, Hals leicht nach
oben. Beide Füße stehen auf dem Boden. Der Hals liegt **nicht** auf dem Oberschenkel auf.

**Im Stehen:** Gurt so einstellen, dass die Gitarre auf derselben Höhe hängt wie im Sitzen. Wer sie
tiefer hängt, weil es besser aussieht, greift schlechter — das ist keine Meinung, das ist Geometrie.

<Handhaltung>
**Greifhand:** Daumen flach hinter dem Hals, ungefähr auf Höhe des Mittelfingers. Nicht über den Hals
gelegt. Die Finger kommen von oben senkrecht auf die Saite, gegriffen wird mit der Fingerkuppe direkt
hinter dem Bundstäbchen — nicht darauf, nicht in der Mitte des Bundfelds. Handgelenk leicht nach vorn
gewölbt, aber nicht geknickt.

**Anschlagshand:** Unterarm liegt auf der Korpuskante auf, das Handgelenk bleibt locker. Die Bewegung
kommt aus dem Handgelenk, nicht aus dem ganzen Arm.
</Handhaltung>

<Fehlerbild>
Der Daumen wandert über den Hals und die Handfläche berührt den Hals. Dann können die Finger nicht
mehr senkrecht stehen, die Nachbarsaiten werden abgedämpft, und der erste Barré-Akkord wird
unmöglich. Prüfe: passt zwischen Handfläche und Hals noch ein Bleistift?
</Fehlerbild>

<Merksatz>Der Daumen ist ein Gegenlager, keine Klemme. Er drückt, er umklammert nicht.</Merksatz>

<Uebung>
Setze die Gitarre dreimal hintereinander neu an — sitzend, stehend, sitzend. Prüfe jedes Mal:
Schultern locker, Daumen hinter dem Hals, Handgelenk nicht geknickt. Fünf Minuten reichen.
</Uebung>

<Hinweis>Wenn nach fünf Minuten etwas zieht, ist es die Haltung, nicht die fehlende Kondition.</Hinweis>

<Checkpoint>Du kannst zwei Minuten sitzen und stehen, ohne dass Schulter, Nacken oder Handgelenk verkrampfen.</Checkpoint>
```

```mdx
export const meta = { id: "07-akkordwechsel", title: "Akkordwechsel", order: 7, requires: "06-offene-akkorde" };

<Ziel>Du wechselst zwischen Em, Am, C und G flüssig, ohne dass der Rhythmus einbricht.</Ziel>

Akkorde zu greifen ist das eine, sie im Takt zu wechseln das andere. Der Trick ist nicht
Geschwindigkeit, sondern Ökonomie: möglichst wenige Finger bewegen.

**Gemeinsame Finger finden.** Zwischen Em und Am bleibt kein Finger stehen, aber die Form wandert
nur eine Saite. Zwischen C und Am bleiben Zeige- und Mittelfinger fast an Ort und Stelle — nur der
Ringfinger springt. Wer das sieht, spart die Hälfte der Bewegung.

<Handhaltung>
Beim Wechsel heben alle beteiligten Finger **gleichzeitig** ab und setzen **gleichzeitig** wieder auf.
Nicht nacheinander. Die Hand behält dabei ihre Form in der Luft — sie fällt nicht auseinander und
baut sich nicht neu auf.
</Handhaltung>

<Fehlerbild>
Finger für Finger aufsetzen. Das klingt beim langsamen Üben sauber und bricht bei jedem Tempo über
70 BPM sofort zusammen. Wenn du deine Finger einzeln landen hörst, übst du den Fehler ein.
</Fehlerbild>

<Merksatz>Langsam sauber schlägt schnell schmutzig. Was du langsam falsch übst, spielst du schnell falsch.</Merksatz>

<Uebung bpm={70}>
**Zielwechsel pro Minute.** Metronom auf 70, eine Minute lang zwischen zwei Akkorden hin und her,
auf jeden vierten Schlag ein Wechsel. Zähle die sauberen Wechsel. Notiere die Zahl.

Paare, in dieser Reihenfolge: Em↔Am, Am↔C, C↔G, G↔D, D↔Am.

Steigere erst das Tempo, wenn ein Paar eine Minute lang fehlerfrei läuft.
</Uebung>

<Checkpoint>Du schaffst Em↔Am eine Minute lang bei 70 BPM ohne einen einzigen scheppernden Wechsel.</Checkpoint>
```

Finger, welcher Bund, welcher Winkel — nie allgemein („locker bleiben").

```mdx
export const meta = { id: "nice-to-know", title: "Nice to know", order: 1, requires: null };

Dinge, die in keinem Kurs vorkommen und trotzdem jeden Tag zählen.

## Saiten

**Saitenstärke.** Die Zahl auf der Packung ist der Durchmesser der dünnsten Saite in Zoll. `.009`
(„Neuner") sind leicht zu greifen und zu ziehen, klingen aber dünner und scheppern eher. `.010`
(„Zehner") sind der übliche Kompromiss. Wer tiefer stimmt, braucht dickere Saiten, sonst wird es
schlabberig.

**Material.** Nickel-Wound klingt weicher, Stainless Steel schärfer und hält länger, kostet aber
Bundstäbchen. Beschichtete Saiten halten zwei- bis dreimal so lange und klingen etwas gedeckter.

<Merksatz>Dünn ist leicht zu spielen, dick ist leicht zu hören. Beides zugleich gibt es nicht.</Merksatz>

**Saitenwechsel.** Alle zwei bis drei Monate, bei viel Spielen öfter. Einzeln wechseln, nicht alle
auf einmal — dann bleibt die Halsspannung konstant. Drei bis vier Wicklungen um den Mechanikposten,
von oben nach unten. Neue Saiten nach dem Aufziehen mehrfach vorsichtig ziehen, sonst verstimmen sie
sich eine Woche lang.

## Einstellung

**Halskrümmung.** Der Hals ist nicht gerade, sondern minimal konkav. Prüfen: erste und letzte Bund
gleichzeitig greifen, in der Mitte sollte ein hauchdünner Spalt bleiben. Korrigiert wird über den
Halsstab — eine Vierteldrehung, dann einen Tag warten.

**Saitenlage.** Der Abstand Saite zu Bund am 12. Bund. Tief spielt sich leicht, schnarrt aber. Hoch
klingt sauberer, kostet Kraft. Ändern erst, wenn die Halskrümmung stimmt.

**Intonation.** Wenn der Ton am 12. Bund höher klingt als das Flageolett an derselben Stelle, ist der
Reiter zu weit vorn. Immer erst stimmen, dann prüfen, dann korrigieren, dann wieder stimmen.

<Merksatz>Erst Hals, dann Saitenlage, dann Intonation. Wer die Reihenfolge dreht, fängt zweimal an.</Merksatz>

## Klang

**Tonabnehmer.** Single Coil klingt hell und drahtig, brummt aber bei Neonlicht und Bildschirmen.
Humbucker hat zwei gegenläufige Spulen — das Brummen hebt sich auf, der Ton wird dicker und mittiger.

**Gain-Struktur.** Verzerrung entsteht dort, wo das Signal zuerst übersteuert. Wenig Gain und
lauterer Amp klingt lebendiger als viel Gain und leiser Amp. Beim Üben zu Hause: weniger Gain als
man denkt — Verzerrung versteckt Fehler, die man hören müsste.

**Effektreihenfolge.** Die übliche Kette: Tuner → Wah → Kompressor → Overdrive/Distortion → Modulation
(Chorus, Phaser) → Delay → Reverb. Alles davor beeinflusst, was danach kommt.

<Merksatz>Verzerrung vor Hall. Wer Hall verzerrt, bekommt Matsch statt Raum.</Merksatz>

## Pflege

**Kabel.** Nie am Kabel ziehen, immer am Stecker. Aufwickeln in gleichmäßigen Schlaufen, nicht um den
Ellbogen. Ein Wackelkontakt kostet mehr Übungszeit als jedes Plateau.

**Griffbrettpflege.** Bei Saitenwechsel mit einem trockenen Tuch abwischen. Palisander ein- bis
zweimal im Jahr ölen, Ahorn mit Lack gar nicht.

## Ohren und Kopf

**Gehörschutz.** Am Verstärker ab Zimmerlautstärke Stöpsel mit linearer Dämpfung. Gehörschäden sind
die einzige Sache an diesem Instrument, die man nicht rückgängig machen kann.

<Merksatz>Ohren sind kein Verschleißteil. Es gibt keine Ersatzohren.</Merksatz>

**Plateau.** Nach einigen Wochen scheint der Fortschritt zu stoppen. Das ist normal: die Hand
automatisiert, was sie kann, bevor sie Neues aufnimmt. Was hilft: Tempo zurücknehmen statt hoch,
etwas anderes üben, und nach ein paar Tagen zurückkommen. Was nicht hilft: länger üben.

**Übungslänge.** Zwanzig Minuten täglich schlagen zwei Stunden am Wochenende deutlich. Das Gehirn
festigt Bewegungen im Schlaf, nicht in der Wiederholung.
```

```tsx
import { useEffect, useState } from "react";
import { useMetronome } from "../../../packages/web/src/audio/useMetronome.js";
import type { WidgetProps } from "../../../packages/web/src/skills/widgets.js";

interface Block {
  id: string;
  titel: string;
  minuten: number;
  beschreibung: string;
  bpm?: number;
}

function format(seconds: number): string {
  const m = String(Math.floor(seconds / 60)).padStart(2, "0");
  const s = String(seconds % 60).padStart(2, "0");
  return `${m}:${s}`;
}

export default function PracticePlan({ data, progress }: WidgetProps) {
  const blocks = (data.flat() as Block[]).filter((block) => block && typeof block.id === "string");
  const [activeId, setActiveId] = useState<string | null>(null);
  const [remaining, setRemaining] = useState(0);
  const metronome = useMetronome();

  useEffect(() => {
    if (activeId === null) return;
    const timer = setInterval(() => setRemaining((value) => Math.max(0, value - 1)), 1000);
    return () => clearInterval(timer);
  }, [activeId]);

  const total = blocks.reduce((sum, block) => sum + block.minuten, 0);
  const active = blocks.find((block) => block.id === activeId);

  function startBlock(block: Block) {
    setActiveId(block.id);
    setRemaining(block.minuten * 60);
    if (block.bpm) metronome.setBpm(block.bpm);
  }

  function finish(block: Block) {
    progress.record(block.id, "correct");
    metronome.stop();
    setActiveId(null);
    setRemaining(0);
  }

  return (
    <div className="space-y-5">
      <div>
        <h1 className="text-2xl font-semibold">Übungsplan</h1>
        <p className="mt-1 text-sm text-slate-600 dark:text-slate-400">
          {total} Minuten insgesamt. Lieber täglich kurz als einmal lang.
        </p>
      </div>

      <ul className="space-y-3">
        {blocks.map((block) => (
          <li key={block.id} className="rounded border border-slate-200 p-4 dark:border-slate-800">
            <div className="flex items-center justify-between">
              <h2 className="font-medium">{block.titel}</h2>
              <span className="text-sm text-slate-500">
                {block.minuten} Minuten{block.bpm ? ` · ${block.bpm} BPM` : ""}
              </span>
            </div>
            <p className="mt-1 text-sm text-slate-600 dark:text-slate-400">{block.beschreibung}</p>

            {activeId === block.id ? (
              <div className="mt-3 flex items-center gap-3">
                <span data-testid="timer" className="font-mono text-2xl">
                  {format(remaining)}
                </span>
                <button
                  type="button"
                  onClick={() => (metronome.running ? metronome.stop() : metronome.start())}
                  className="rounded border border-slate-300 px-3 py-1 text-sm dark:border-slate-700"
                >
                  Metronom {metronome.running ? "aus" : "an"} ({metronome.bpm})
                </button>
                <button
                  type="button"
                  onClick={() => finish(block)}
                  className="rounded bg-slate-900 px-3 py-1 text-sm text-white dark:bg-slate-100 dark:text-slate-900"
                >
                  Fertig
                </button>
              </div>
            ) : (
              <button
                type="button"
                onClick={() => startBlock(block)}
                className="mt-3 rounded border border-slate-300 px-3 py-1 text-sm dark:border-slate-700"
              >
                Starten
              </button>
            )}
          </li>
        ))}
      </ul>

      {active ? <p className="text-sm text-slate-500">Läuft: {active.titel}</p> : null}
    </div>
  );
}
```

```tsx
function LineWithChords({ line }: { line: Song["sections"][number]["lines"][number] }) {
  if (line.chords.length === 0) {
    return <div className="whitespace-pre font-mono text-sm">{line.lyrics}</div>;
  }

  let chordRow = "";
  for (const chord of line.chords) {
    if (chord.position > chordRow.length) chordRow = chordRow.padEnd(chord.position, " ");
    chordRow += chord.symbol;
  }

  return (
    <div className="font-mono text-sm">
      <div className="whitespace-pre font-medium text-amber-700 dark:text-amber-400">{chordRow}</div>
      <div className="whitespace-pre">{line.lyrics}</div>
    </div>
  );
}

export function SongView({ song }: { song: Song }) {
```

```ts
import * as cheerio from "cheerio";

export type FetchErrorCode =
  | "bad_url"
  | "blocked_host"
  | "timeout"
  | "too_large"
  | "unreachable"
  | "no_content";

export class FetchError extends Error {
  readonly code: FetchErrorCode;
  constructor(code: FetchErrorCode, message: string) {
    super(message);
    this.name = "FetchError";
    this.code = code;
  }
}

const PRIVATE_V4 = [
  /^127\./,
  /^10\./,
  /^192\.168\./,
  /^169\.254\./,
  /^172\.(1[6-9]|2\d|3[01])\./,
  /^0\./,
];

function isBlockedHost(hostname: string): boolean {
  const host = hostname.replace(/^\[|\]$/g, "").toLowerCase();
  if (host === "localhost" || host.endsWith(".localhost") || host.endsWith(".local")) return true;
  if (host === "::1" || host === "0:0:0:0:0:0:0:1") return true;
  if (/^f[cd][0-9a-f]{2}:/.test(host)) return true;
  if (/^fe80:/.test(host)) return true;
  return PRIVATE_V4.some((pattern) => pattern.test(host));
}

export function assertPublicUrl(input: string): URL {
  let url: URL;
  try {
    url = new URL(input);
  } catch {
    throw new FetchError("bad_url", "Das ist keine gültige URL.");
  }

  if (url.protocol !== "http:" && url.protocol !== "https:") {
    throw new FetchError("bad_url", "Nur http- und https-Adressen werden abgerufen.");
  }

  if (isBlockedHost(url.hostname)) {
    throw new FetchError(
      "blocked_host",
      "Diese Adresse zeigt auf das lokale Netz und wird nicht abgerufen.",
    );
  }

  return url;
}

export function extractSongText(html: string): string {
  const $ = cheerio.load(html);
  $("script, style, noscript").remove();

  const pres = $("pre")
    .map((_, element) => $(element).text())
    .get()
    .map((text) => text.trim())
    .filter((text) => text.length > 0);

  const candidate =
    pres.length > 0
      ? pres.join("\n\n")
      : $("body")
          .text()
          .split(/\r?\n/)
          .map((line) => line.trimEnd())
          .join("\n")
          .trim();

  if (candidate.trim().length === 0) {
    throw new FetchError("no_content", "Auf der Seite wurde kein verwertbarer Inhalt gefunden.");
  }

  return candidate;
}

interface FetchOptions {
  timeoutMs?: number;
  maxBytes?: number;
  fetchImpl?: typeof fetch;
}

const MAX_REDIRECTS = 3;

export async function fetchSongFromUrl(
  input: string,
  options: FetchOptions = {},
): Promise<{ text: string; title: string | null }> {
  const timeoutMs = options.timeoutMs ?? 10_000;
  const maxBytes = options.maxBytes ?? 2_000_000;
  const doFetch = options.fetchImpl ?? fetch;

  let url = assertPublicUrl(input);
  let response: Response | null = null;

  for (let hop = 0; hop <= MAX_REDIRECTS; hop += 1) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeoutMs);

    let current: Response;
    try {
      current = await doFetch(url.toString(), {
        redirect: "manual",
        signal: controller.signal,
        headers: { "user-agent": "SkillsApp/1.0 (persönlicher Gebrauch)", accept: "text/html,text/plain" },
      });
    } catch (error) {
      clearTimeout(timer);
      if (error instanceof Error && error.name === "AbortError") {
        throw new FetchError("timeout", "Die Seite hat zu lange gebraucht. Füge den Text stattdessen ein.");
      }
      throw new FetchError(
        "unreachable",
        "Die Seite konnte nicht geladen werden. Kopiere den Text und füge ihn stattdessen ein.",
      );
    } finally {
      clearTimeout(timer);
    }

    if (current.status >= 300 && current.status < 400) {
      const location = current.headers.get("location");
      if (!location) {
        throw new FetchError("unreachable", "Die Seite hat eine unvollständige Weiterleitung geschickt.");
      }
      url = assertPublicUrl(new URL(location, url).toString());
      continue;
    }

    response = current;
    break;
  }

  if (!response) {
    throw new FetchError("unreachable", "Zu viele Weiterleitungen. Füge den Text stattdessen ein.");
  }

  if (!response.ok) {
    throw new FetchError("unreachable", `Die Seite antwortete mit Status ${response.status}.`);
  }

  const declared = Number(response.headers.get("content-length") ?? "0");
  if (declared > maxBytes) {
    throw new FetchError("too_large", "Die Seite ist zu groß. Füge den Text stattdessen ein.");
  }

  const html = await response.text();
  if (html.length > maxBytes) {
    throw new FetchError("too_large", "Die Seite ist zu groß. Füge den Text stattdessen ein.");
  }

  const title = /<title[^>]*>([^<]*)<\/title>/i.exec(html)?.[1]?.trim() ?? null;
  return { text: extractSongText(html), title: title && title.length > 0 ? title : null };
}
```

```ts
import { detectFormat, parseSong } from "@skills/core";
import { Hono } from "hono";
import { z } from "zod";
import type { AppEnv } from "../app.js";
import { assertPublicUrl, fetchSongFromUrl, FetchError } from "../services/songFetch.js";

const createSchema = z.object({
  content: z.string().min(1),
  title: z.string().optional(),
  artist: z.string().optional(),
  sourceUrl: z.string().optional(),
  difficulty: z.number().int().min(1).max(5).optional(),
  tags: z.array(z.string()).default([]),
});

export const songRoutes = new Hono<AppEnv>();

songRoutes.get("/api/songs", (c) => {
  const songs = c
    .get("database")
    .raw.prepare(
      "SELECT id, title, artist, source_url AS sourceUrl, format, content, added_at AS addedAt, difficulty, tags FROM songs ORDER BY added_at DESC",
    )
    .all();
  return c.json({ songs });
});

songRoutes.get("/api/songs/:id", (c) => {
  const song = c
    .get("database")
    .raw.prepare(
      "SELECT id, title, artist, source_url AS sourceUrl, format, content, added_at AS addedAt, difficulty, tags FROM songs WHERE id = ?",
    )
    .get(Number(c.req.param("id")));
  if (!song) return c.json({ error: { code: "not_found", message: "Song nicht gefunden." } }, 404);
  return c.json({ song });
});

songRoutes.post("/api/songs", async (c) => {
  const body = await c.req.json().catch(() => null);
  const parsed = createSchema.safeParse(body);
  if (!parsed.success || parsed.data.content.trim() === "") {
    return c.json(
      { error: { code: "invalid_body", message: "Der Songtext darf nicht leer sein." } },
      400,
    );
  }

  const song = parseSong(parsed.data.content);
  const format = detectFormat(parsed.data.content);
  const title = parsed.data.title?.trim() || song.title;
  const artist = parsed.data.artist?.trim() || song.artist;
  const now = Date.now();

  const info = c
    .get("database")
    .raw.prepare(
      `INSERT INTO songs (title, artist, source_url, format, content, added_at, difficulty, tags)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
    )
    .run(
      title,
      artist,
      parsed.data.sourceUrl ?? null,
      format,
      parsed.data.content,
      now,
      parsed.data.difficulty ?? null,
      JSON.stringify(parsed.data.tags),
    );

  return c.json({
    song: {
      id: Number(info.lastInsertRowid),
      title,
      artist,
      format,
      content: parsed.data.content,
      addedAt: now,
    },
  });
});

songRoutes.delete("/api/songs/:id", (c) => {
  c.get("database").raw.prepare("DELETE FROM songs WHERE id = ?").run(Number(c.req.param("id")));
  return c.json({ ok: true });
});

songRoutes.post("/api/songs/fetch", async (c) => {
  const body = await c.req.json().catch(() => null);
  const parsed = z.object({ url: z.string().min(1) }).safeParse(body);
  if (!parsed.success) {
    return c.json({ error: { code: "invalid_body", message: "Es fehlt eine URL." } }, 400);
  }

  try {
    assertPublicUrl(parsed.data.url);
    const result = await fetchSongFromUrl(parsed.data.url);
    return c.json(result);
  } catch (error) {
    if (error instanceof FetchError) {
      return c.json({ error: { code: error.code, message: error.message } }, 400);
    }
    throw error;
  }
});
```

```tsx
import { detectFormat, parseSong } from "@skills/core";
import { useMutation } from "@tanstack/react-query";
import { useMemo, useState } from "react";
import { useNavigate } from "react-router-dom";
import { api, ApiError } from "../../api/client.js";
import { SongView } from "./SongView.js";

type Mode = "paste" | "url";

const FORMAT_LABELS: Record<string, string> = {
  chordpro: "ChordPro",
  chordsheet: "Akkorde über Text",
  tab: "ASCII-Tabulatur",
};

export function SongImport() {
  const navigate = useNavigate();
  const [mode, setMode] = useState<Mode>("paste");
  const [url, setUrl] = useState("");
  const [text, setText] = useState("");
  const [title, setTitle] = useState("");
  const [artist, setArtist] = useState("");

  const parsed = useMemo(() => (text.trim() === "" ? null : parseSong(text)), [text]);
  const format = useMemo(() => (text.trim() === "" ? null : detectFormat(text)), [text]);

  const fetchUrl = useMutation({
    mutationFn: () => api.post<{ text: string; title: string | null }>("/api/songs/fetch", { url }),
    onSuccess: (result) => {
      setText(result.text);
      const song = parseSong(result.text);
      setTitle(result.title ?? song.title);
      setArtist(song.artist ?? "");
    },
  });

  const save = useMutation({
    mutationFn: () =>
      api.post<{ song: { id: number } }>("/api/songs", {
        content: text,
        title: title.trim() || undefined,
        artist: artist.trim() || undefined,
        sourceUrl: mode === "url" && url.trim() !== "" ? url : undefined,
        tags: [],
      }),
    onSuccess: () => navigate("/skill/guitar/songs"),
  });

  function onTextChange(value: string) {
    setText(value);
    if (value.trim() === "") return;
    const song = parseSong(value);
    if (title.trim() === "") setTitle(song.title);
    if (artist.trim() === "" && song.artist) setArtist(song.artist);
  }

  const error = fetchUrl.error ?? save.error;

  return (
    <div className="max-w-3xl space-y-6">
      <h1 className="text-2xl font-semibold">Song hinzufügen</h1>

      <div className="flex gap-4 text-sm">
        <label className="flex items-center gap-2">
          <input type="radio" checked={mode === "paste"} onChange={() => setMode("paste")} />
          <span>Text einfügen</span>
        </label>
        <label className="flex items-center gap-2">
          <input type="radio" checked={mode === "url"} onChange={() => setMode("url")} />
          <span>URL abrufen</span>
        </label>
      </div>

      {mode === "url" ? (
        <div className="flex gap-2">
          <label className="flex-1 text-sm">
            <span className="mb-1 block">Adresse</span>
            <input
              value={url}
              onChange={(e) => setUrl(e.target.value)}
              placeholder="https://…"
              className="w-full rounded border border-slate-300 px-3 py-2 dark:border-slate-700 dark:bg-slate-900"
            />
          </label>
          <button
            type="button"
            disabled={url.trim() === "" || fetchUrl.isPending}
            onClick={() => fetchUrl.mutate()}
            className="self-end rounded bg-slate-900 px-4 py-2 text-white disabled:opacity-50 dark:bg-slate-100 dark:text-slate-900"
          >
            {fetchUrl.isPending ? "Wird geholt…" : "Abrufen"}
          </button>
        </div>
      ) : null}

      {error ? (
        <p className="rounded border border-amber-300 bg-amber-50 p-3 text-sm text-amber-800 dark:border-amber-900 dark:bg-amber-950 dark:text-amber-200">
          {error instanceof ApiError ? error.message : "Unbekannter Fehler."}
        </p>
      ) : null}

      <label className="block text-sm">
        <span className="mb-1 block">Songtext</span>
        <textarea
          value={text}
          onChange={(e) => onTextChange(e.target.value)}
          rows={10}
          placeholder="Akkorde und Text hier einfügen — ChordPro, Akkorde über Text oder ASCII-Tab."
          className="w-full rounded border border-slate-300 px-3 py-2 font-mono text-sm dark:border-slate-700 dark:bg-slate-900"
        />
      </label>

      {format ? (
        <p className="text-sm text-slate-600 dark:text-slate-400">
          Erkanntes Format: <strong>{FORMAT_LABELS[format]}</strong>
        </p>
      ) : null}

      {parsed ? (
        <>
          <div className="grid gap-3 sm:grid-cols-2">
            <label className="text-sm">
              <span className="mb-1 block">Titel</span>
              <input
                value={title}
                onChange={(e) => setTitle(e.target.value)}
                className="w-full rounded border border-slate-300 px-3 py-2 dark:border-slate-700 dark:bg-slate-900"
              />
            </label>
            <label className="text-sm">
              <span className="mb-1 block">Interpret</span>
              <input
                value={artist}
                onChange={(e) => setArtist(e.target.value)}
                className="w-full rounded border border-slate-300 px-3 py-2 dark:border-slate-700 dark:bg-slate-900"
              />
            </label>
          </div>

          <SongView song={parsed} />

          <button
            type="button"
            disabled={save.isPending}
            onClick={() => save.mutate()}
            className="rounded bg-slate-900 px-4 py-2 text-white disabled:opacity-50 dark:bg-slate-100 dark:text-slate-900"
          >
            {save.isPending ? "Wird gespeichert…" : "Speichern"}
          </button>
        </>
      ) : null}
    </div>
  );
}
```

```markdown
## Audio in einem Widget

Zwei Hooks stehen bereit und kapseln allen Gerätezugriff:

- `useMicrophone({ onBuffer })` liefert Audiopuffer und meldet Zustände `idle`, `requesting`,
```
