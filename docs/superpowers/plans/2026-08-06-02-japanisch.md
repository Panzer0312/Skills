**Goal:** Ein vollständiger Lernbereich Japanisch — Kana-Schnellzugriff, Kana-Kurs, Themen-Vokabeln, Grammatik und ein eigenes Lese-Modul — plus Import bestehender Anki-Decks in denselben Scheduler.

**Tech Stack:** Zusätzlich zu Plan 1: `fflate` (Zip lesen und schreiben, keine nativen Abhängigkeiten), `better-sqlite3` liest die eingebettete Anki-Datenbank.

## Global Constraints

- Voraussetzung: Plan 1 ist abgeschlossen, `npm test` ist grün.

```ts
const BASE: Record<string, string> = {
  あ: "a", い: "i", う: "u", え: "e", お: "o",
  か: "ka", き: "ki", く: "ku", け: "ke", こ: "ko",

  ま: "ma", み: "mi", む: "mu", め: "me", も: "mo",
  や: "ya", ゆ: "yu", よ: "yo",
  ら: "ra", り: "ri", る: "ru", れ: "re", ろ: "ro",
  わ: "wa", を: "wo", ん: "n",
  ゃ: "ya", ゅ: "yu", ょ: "yo", ぁ: "a", ぃ: "i", ぅ: "u", ぇ: "e", ぉ: "o",
};

const YOON_STEM: Record<string, string> = {
  き: "ky", ぎ: "gy", し: "sh", じ: "j", ち: "ch", ぢ: "j", に: "ny",
  ひ: "hy", び: "by", ぴ: "py", み: "my", り: "ry",
};

const SMALL_Y: Record<string, string> = { ゃ: "a", ゅ: "u", ょ: "o" };

const HIRAGANA_START = 0x3041;
const HIRAGANA_END = 0x3096;
const KATAKANA_START = 0x30a1;
const KATAKANA_END = 0x30f6;
const SHIFT = KATAKANA_START - HIRAGANA_START;

export function isKana(char: string): boolean {
  const code = char.codePointAt(0);
  if (code === undefined) return false;
  return (
    (code >= HIRAGANA_START && code <= HIRAGANA_END) ||
    (code >= KATAKANA_START && code <= KATAKANA_END) ||
    char === "ー"
  );
}

export function toHiragana(input: string): string {
  return [...input]
    .map((char) => {
      const code = char.codePointAt(0)!;
      return code >= KATAKANA_START && code <= KATAKANA_END
        ? String.fromCodePoint(code - SHIFT)
        : char;
    })
    .join("");
}

export function toKatakana(input: string): string {
  return [...input]
    .map((char) => {
      const code = char.codePointAt(0)!;
      return code >= HIRAGANA_START && code <= HIRAGANA_END
        ? String.fromCodePoint(code + SHIFT)
        : char;
    })
    .join("");
}

export function kanaToRomaji(input: string, options: { particles?: boolean } = {}): string {
  const chars = [...toHiragana(input)];
  const out: string[] = [];
  let pendingGemination = false;

  for (let i = 0; i < chars.length; i += 1) {
    const char = chars[i]!;

    if (char === "っ") {
      pendingGemination = true;
      continue;
    }

    if (char === "ー") {
      const last = out[out.length - 1];
      if (last) out.push(last[last.length - 1]!);
      continue;
    }

    let syllable: string | null = null;
    const next = chars[i + 1];

    if (next && SMALL_Y[next] && YOON_STEM[char]) {
      syllable = `${YOON_STEM[char]}${SMALL_Y[next]}`;
      i += 1;
    } else if (BASE[char]) {
      syllable = BASE[char]!;
      if (options.particles) {
        if (char === "は" && i > 0) syllable = "wa";
        else if (char === "へ" && i > 0) syllable = "e";
        else if (char === "を") syllable = "o";
      }
    }

    if (syllable === null) {
      out.push(char);
      pendingGemination = false;
      continue;
    }

    if (pendingGemination) {
      syllable = `${syllable[0]}${syllable}`;
      pendingGemination = false;
    }

    out.push(syllable);
  }

  return out.join("");
}

const REVERSE_LOOKUP: [string, string][] = [];
for (const [stem, prefix] of Object.entries(YOON_STEM)) {
  for (const [small, vowel] of Object.entries(SMALL_Y)) {
    REVERSE_LOOKUP.push([`${prefix}${vowel}`, `${stem}${small}`]);
  }
}
for (const [kana, romaji] of Object.entries(BASE)) {
  REVERSE_LOOKUP.push([romaji, kana]);
}
REVERSE_LOOKUP.sort((a, b) => b[0].length - a[0].length);

export function romajiToKana(input: string, script: "hiragana" | "katakana"): string {
  let rest = input.toLowerCase();
  let out = "";

  while (rest.length > 0) {
    const doubled = /^([kstpgzdbhfmyrwjc])\1/.exec(rest);
    if (doubled) {
      out += "っ";
      rest = rest.slice(1);
      continue;
    }

    const previousVowel = /[aeiou]$/.exec(out.length > 0 ? kanaToRomaji(out) : "");
    if (previousVowel && rest.startsWith(previousVowel[0]) && script === "katakana") {
      out += "ー";
      rest = rest.slice(1);
      continue;
    }

    const match = REVERSE_LOOKUP.find(([romaji]) => rest.startsWith(romaji));
    if (!match) {
      out += rest[0];
      rest = rest.slice(1);
      continue;
    }
    out += match[1];
    rest = rest.slice(match[0].length);
  }

  return script === "katakana" ? toKatakana(out) : out;
}
```

```ts
import { z } from "zod";

export const kanaEntrySchema = z.object({
  kana: z.string().min(1),
  romaji: z.string().min(1),
  strokes: z.number().int().positive(),
  merksatz: z.string().min(10),
  beispiel: z.object({
    wort: z.string().min(1),
    lesung: z.string().min(1),
    bedeutung: z.string().min(1),
  }),
  verwechslung: z.array(z.string()).default([]),
});

export const kanaFileSchema = z.object({
  script: z.enum(["hiragana", "katakana"]),
  entries: z.array(kanaEntrySchema).min(1),
});

export const kanjiEntrySchema = z.object({
  kanji: z.string().min(1),
  bedeutung: z.string().min(1),
  on: z.array(z.string()).default([]),
  kun: z.array(z.string()).default([]),
  strokes: z.number().int().positive(),
  radikal: z.string().min(1),
  merksatz: z.string().min(10),
  beispiele: z
    .array(
      z.object({
        wort: z.string().min(1),
        lesung: z.string().min(1),
        bedeutung: z.string().min(1),
        lesungsart: z.enum(["on", "kun"]),
      }),
    )
    .min(1),
});

export const satzSchema = z.object({
  id: z.string().min(1),
  japanisch: z.string().min(1),
  stufe: z.enum(["wort", "satz", "absatz"]),
  uebersetzung: z.string().min(1),
  tokens: z
    .array(
      z.object({
        text: z.string().min(1),
        lesung: z.string().optional(),
        bedeutung: z.string().optional(),
        begruendung: z.string().optional(),
      }),
    )
    .min(1),
});

export type KanaEntry = z.infer<typeof kanaEntrySchema>;
export type KanaFile = z.infer<typeof kanaFileSchema>;
export type KanjiEntry = z.infer<typeof kanjiEntrySchema>;
export type Satz = z.infer<typeof satzSchema>;

function fail(source: string, error: z.ZodError): { ok: false; error: string } {
  return {
    ok: false,
    error: `${source} — ${error.issues.map((i) => `${i.path.join(".")}: ${i.message}`).join("; ")}`,
  };
}

export function parseKanaFile(
  input: unknown,
  source: string,
): { ok: true; file: KanaFile } | { ok: false; error: string } {
  const parsed = kanaFileSchema.safeParse(input);
  if (!parsed.success) return fail(source, parsed.error);
  const seen = new Set<string>();
  for (const entry of parsed.data.entries) {
    if (seen.has(entry.kana)) {
      return { ok: false, error: `${source} — Zeichen "${entry.kana}" kommt mehrfach vor.` };
    }
    seen.add(entry.kana);
  }
  return { ok: true, file: parsed.data };
}

export function parseKanjiFile(
  input: unknown,
  source: string,
): { ok: true; entries: KanjiEntry[] } | { ok: false; error: string } {
  const parsed = z.array(kanjiEntrySchema).safeParse(input);
  if (!parsed.success) return fail(source, parsed.error);
  return { ok: true, entries: parsed.data };
}

export function parseSatzFile(
  input: unknown,
  source: string,
): { ok: true; saetze: Satz[] } | { ok: false; error: string } {
  const parsed = z.array(satzSchema).safeParse(input);
  if (!parsed.success) return fail(source, parsed.error);
  return { ok: true, saetze: parsed.data };
}
```

```tsx
import { parseKanaFile, type KanaEntry } from "@skills/core";
import { useCallback, useMemo, useState } from "react";
import type { WidgetProps } from "../../../packages/web/src/skills/widgets.js";

type Direction = "kana-to-romaji" | "romaji-to-kana";

function pick(entries: KanaEntry[], weight: (entry: KanaEntry) => number): KanaEntry {
  const total = entries.reduce((sum, entry) => sum + weight(entry), 0);
  let cursor = Math.random() * total;
  for (const entry of entries) {
    cursor -= weight(entry);
    if (cursor <= 0) return entry;
  }
  return entries[entries.length - 1]!;
}

export default function KanaDrill({ data, progress }: WidgetProps) {
  const entries = useMemo(
    () =>
      data.flatMap((file) => {
        const parsed = parseKanaFile(file, "kana");
        return parsed.ok ? parsed.file.entries : [];
      }),
    [data],
  );

  const weight = useCallback(
    (entry: KanaEntry) => (progress.get(entry.kana) === "wrong" ? 3 : 1),
    [progress],
  );

  const [direction, setDirection] = useState<Direction>("kana-to-romaji");
  const [current, setCurrent] = useState<KanaEntry>(() => pick(entries, weight));
  const [answer, setAnswer] = useState("");
  const [verdict, setVerdict] = useState<"correct" | "wrong" | null>(null);

  if (entries.length === 0) {
    return <p className="text-red-600 dark:text-red-400">Keine Kana-Daten geladen.</p>;
  }

  const expected = direction === "kana-to-romaji" ? current.romaji : current.kana;
  const prompt = direction === "kana-to-romaji" ? current.kana : current.romaji;

  function check() {
    const correct = answer.trim().toLowerCase() === expected.toLowerCase();
    setVerdict(correct ? "correct" : "wrong");
    progress.record(current.kana, correct ? "correct" : "wrong");
  }

  function next() {
    setCurrent(pick(entries, weight));
    setAnswer("");
    setVerdict(null);
  }

  return (
    <div className="mx-auto max-w-md space-y-6">
      <div className="flex gap-2">
        {(
          [
            ["kana-to-romaji", "Zeichen → Laut"],
            ["romaji-to-kana", "Laut → Zeichen"],
          ] as const
        ).map(([value, label]) => (
          <button
            key={value}
            type="button"
            onClick={() => {
              setDirection(value);
              next();
            }}
            className={`rounded border px-3 py-1 text-sm ${
              direction === value
                ? "border-slate-900 dark:border-slate-100"
                : "border-slate-300 dark:border-slate-700"
            }`}
          >
            {label}
          </button>
        ))}
      </div>

      <div className="rounded-lg border border-slate-200 p-10 text-center dark:border-slate-800">
        <div data-testid="prompt" className="text-6xl">
          {prompt}
        </div>
      </div>

      <input
        type="text"
        value={answer}
        onChange={(e) => setAnswer(e.target.value)}
        onKeyDown={(e) => {
          if (e.key !== "Enter") return;
          if (verdict === null) check();
          else next();
        }}
        className="w-full rounded border border-slate-300 px-3 py-2 text-center text-lg dark:border-slate-700 dark:bg-slate-900"
        placeholder={direction === "kana-to-romaji" ? "Lesung eingeben" : "Zeichen eingeben"}
      />

      {verdict === null ? (
        <button
          type="button"
          onClick={check}
          className="w-full rounded bg-slate-900 px-4 py-2 text-white dark:bg-slate-100 dark:text-slate-900"
        >
          Prüfen
        </button>
      ) : (
        <div className="space-y-3">
          <p className={verdict === "correct" ? "text-emerald-600" : "text-red-600"}>
            {verdict === "correct" ? "Richtig." : `Falsch — richtig wäre: ${expected}`}
          </p>
          {verdict === "wrong" ? (
            <p className="rounded bg-amber-50 p-3 text-sm dark:bg-amber-950">
              <strong>Merksatz: </strong>
              {current.merksatz}
            </p>
          ) : null}
          <button
            type="button"
            onClick={next}
            className="w-full rounded border border-slate-300 px-4 py-2 dark:border-slate-700"
          >
            Weiter
          </button>
        </div>
      )}
    </div>
  );
}
```

```mdx
export const meta = { id: "02-partikel-wa-ga", title: "は und が", order: 2, requires: "01-satzbau" };

<Ziel>Du weißt, wann は und wann が steht, und hörst den Unterschied im Satz.</Ziel>

Beide Partikeln stehen hinter dem, worüber gesprochen wird — aber sie tun Verschiedenes.
は hebt ein **Thema** hervor: „was X angeht". が benennt das **Subjekt**, oft neu oder betont.

わたし**は** がくせいです。 — Was mich angeht: Ich bin Student.
だれ**が** きましたか。 — Wer ist gekommen?

<Merksatz>は ist der Rahmen, が ist der Finger. は spannt auf, worüber geredet wird; が zeigt auf den, der handelt.</Merksatz>

Zwei feste Regeln, die fast immer greifen:

1. Nach einem Fragewort steht **が**, nie は. Man kann nicht als Thema setzen, was man noch nicht kennt.
2. In der Antwort auf eine solche Frage steht ebenfalls が.

<Fehlerbild>
Deutschsprachige setzen は überall hin, weil es sich wie „ist" anfühlt. は ist aber keine Kopula —
das ist です. „わたしはです" ist kein Satz.
</Fehlerbild>

<Hinweis>Geschrieben wird は, gesprochen wird es als Partikel **wa**. Siehe „Sonderfälle beim Lesen".</Hinweis>

<Uebung>
Setze は oder が ein:
1. だれ＿ たべましたか。
2. わたし＿ たべました。
3. これ＿ ほんです。

Lösungen: 1. が — 2. が — 3. は
</Uebung>

<Checkpoint>Du kannst erklären, warum nach einem Fragewort nie は steht.</Checkpoint>
```

```tsx
import { parseKanjiFile, parseSatzFile, type KanjiEntry, type Satz } from "@skills/core";
import { useMemo, useState } from "react";
import type { WidgetProps } from "../../../packages/web/src/skills/widgets.js";

type Tab = "kanji" | "regeln" | "training";

interface Regel {
  Regel: string;
  Erklaerung: string;
  Merksatz: string;
  Beispiel: string;
}

export default function ReadingTrainer({ data, progress }: WidgetProps) {
  const [tab, setTab] = useState<Tab>("kanji");

  const kanji = useMemo<KanjiEntry[]>(() => {
    for (const entry of data) {
      const parsed = parseKanjiFile(entry, "kanji");
      if (parsed.ok) return parsed.entries;
    }
    return [];
  }, [data]);

  const saetze = useMemo<Satz[]>(() => {
    for (const entry of data) {
      const parsed = parseSatzFile(entry, "saetze");
      if (parsed.ok) return parsed.saetze;
    }
    return [];
  }, [data]);

  const regeln = useMemo<Regel[]>(
    () =>
      (data.find(
        (entry): entry is Regel[] =>
          Array.isArray(entry) && entry.length > 0 && typeof entry[0] === "object" &&
          entry[0] !== null && "Regel" in (entry[0] as object),
      ) ?? []) as Regel[],
    [data],
  );

  return (
    <div className="space-y-6">
      <div role="tablist" className="flex gap-2">
        {(
          [
            ["kanji", "Kanji"],
            ["regeln", "Regeln"],
            ["training", "Lesetraining"],
          ] as const
        ).map(([value, label]) => (
          <button
            key={value}
            role="tab"
            aria-selected={tab === value}
            type="button"
            onClick={() => setTab(value)}
            className={`rounded border px-3 py-1 text-sm ${
              tab === value ? "border-slate-900 dark:border-slate-100" : "border-slate-300 dark:border-slate-700"
            }`}
          >
            {label}
          </button>
        ))}
      </div>

      {tab === "kanji" ? <KanjiList entries={kanji} /> : null}
      {tab === "regeln" ? <RuleList regeln={regeln} /> : null}
      {tab === "training" ? <Training saetze={saetze} progress={progress} /> : null}
    </div>
  );
}

function KanjiList({ entries }: { entries: KanjiEntry[] }) {
  return (
    <ul className="grid gap-3 sm:grid-cols-2">
      {entries.map((entry) => (
        <li key={entry.kanji} className="rounded border border-slate-200 p-4 dark:border-slate-800">
          <div className="flex items-baseline gap-3">
            <span className="text-4xl">{entry.kanji}</span>
            <span>{entry.bedeutung}</span>
            <span className="ml-auto text-xs text-slate-500">{entry.strokes} Striche</span>
          </div>
          <dl className="mt-2 text-sm">
            <div className="flex gap-2">
              <dt className="text-slate-500">on:</dt>
              <dd>{entry.on.join("、") || "—"}</dd>
            </div>
            <div className="flex gap-2">
              <dt className="text-slate-500">kun:</dt>
              <dd>{entry.kun.join("、") || "—"}</dd>
            </div>
          </dl>
          <p className="mt-2 rounded bg-amber-50 p-2 text-sm dark:bg-amber-950">{entry.merksatz}</p>
          <ul className="mt-2 space-y-1 text-sm text-slate-600 dark:text-slate-400">
            {entry.beispiele.map((b) => (
              <li key={b.wort}>
                {b.wort} ({b.lesung}) — {b.bedeutung} <span className="text-xs">[{b.lesungsart}]</span>
              </li>
            ))}
          </ul>
        </li>
      ))}
    </ul>
```

Hintergrund zum Format: Eine `.apkg`-Datei ist ein Zip. Darin liegt die Sammlung als SQLite — `collection.anki21` bei neueren Exporten, `collection.anki2` bei älteren. Eine Datei `media` enthält eine JSON-Zuordnung von laufender Nummer auf Dateinamen; die Mediendateien selbst liegen unter genau diesen Nummern. Notizfelder stehen in `notes.flds`, getrennt durch `\x1f`. Notiztypen stehen entweder als JSON in `col.models` (altes Schema) oder in eigenen Tabellen `notetypes`, `fields`, `templates` (Schema 18 und neuer). Beides muss unterstützt werden.

Nicht unterstützt wird `collection.anki21b` — dort ist die Datenbank zstd-komprimiert. In dem Fall wird ein klarer Fehler geworfen mit dem Hinweis, im Anki-Export „Unterstützung für ältere Anki-Versionen" zu aktivieren.

```ts
/** Builds a real .apkg byte stream so the parser is tested against the actual format. */
export function buildTestApkg(options: TestDeckOptions): Uint8Array {
  const deckName = options.deckName ?? "Testdeck";
  const fieldNames = options.fieldNames ?? ["Vorderseite", "Rückseite"];
  const templates = options.templates ?? ["Karte 1"];
  const legacy = options.legacySchema ?? true;
  const noteTypeId = 1_600_000_000_000;
  const deckId = 1_600_000_000_001;

  const dir = mkdtempSync(join(tmpdir(), "apkg-build-"));
  const dbPath = join(dir, "collection.db");
  const db = new Database(dbPath);

  db.exec(OLD_SCHEMA);
  if (!legacy) db.exec(NEW_TABLES);

  const models = legacy
    ? JSON.stringify({
        [String(noteTypeId)]: {
          id: noteTypeId,
          name: "Basis",
          flds: fieldNames.map((name, ord) => ({ name, ord })),
          tmpls: templates.map((name, ord) => ({ name, ord })),
        },
      })
    : "{}";

  const decks = legacy
    ? JSON.stringify({ [String(deckId)]: { id: deckId, name: deckName } })
    : "{}";

  db.prepare(
    "INSERT INTO col (id, crt, mod, scm, ver, dty, usn, ls, conf, models, decks, dconf, tags) VALUES (1, 0, 0, 0, ?, 0, 0, 0, '{}', ?, ?, '{}', '{}')",
  ).run(legacy ? 11 : 18, models, decks);

  if (!legacy) {
    db.prepare("INSERT INTO notetypes (id, name, mtime_secs, usn, config) VALUES (?, 'Basis', 0, 0, x'')").run(noteTypeId);
    fieldNames.forEach((name, ord) => {
      db.prepare("INSERT INTO fields (ntid, ord, name, config) VALUES (?, ?, ?, x'')").run(noteTypeId, ord, name);
    });
    templates.forEach((name, ord) => {
      db.prepare(
        "INSERT INTO templates (ntid, ord, name, mtime_secs, usn, config) VALUES (?, ?, ?, 0, 0, x'')",
      ).run(noteTypeId, ord, name);
    });
    db.prepare("INSERT INTO decks (id, name, mtime_secs, usn, common, kind) VALUES (?, ?, 0, 0, x'', x'')").run(
      deckId,
      deckName,
    );
  }

  const insertNote = db.prepare(
    "INSERT INTO notes (id, guid, mid, mod, usn, tags, flds, sfld, csum, flags, data) VALUES (?, ?, ?, 0, 0, ?, ?, ?, 0, 0, '')",
  );
  const insertCard = db.prepare(
    `INSERT INTO cards (id, nid, did, ord, mod, usn, type, queue, due, ivl, factor, reps, lapses, left, odue, odid, flags, data)
     VALUES (?, ?, ?, ?, 0, 0, 2, 2, 5, 10, 2500, 3, 1, 0, 0, 0, 0, '')`,
  );

  options.notes.forEach((note, i) => {
    const noteId = 1_700_000_000_000 + i;
    insertNote.run(
      noteId,
      `guid-${i}`,
      noteTypeId,
      ` ${(note.tags ?? []).join(" ")} `,
      note.fields.join(String.fromCharCode(0x1f)),
      note.fields[0] ?? "",
    );
    templates.forEach((_, ord) => {
      insertCard.run(noteId * 10 + ord, noteId, deckId, ord);
    });
  });

  db.close();

  const collectionName = options.collectionName ?? (legacy ? "collection.anki2" : "collection.anki21");
  const files: Record<string, Uint8Array> = {
    [collectionName]: new Uint8Array(readFileSync(dbPath)),
  };

  const mediaEntries = Object.entries(options.media ?? {});
  const mediaMap: Record<string, string> = {};
  mediaEntries.forEach(([filename, content], i) => {
    mediaMap[String(i)] = filename;
    files[String(i)] = new TextEncoder().encode(content);
  });
  files["media"] = new TextEncoder().encode(JSON.stringify(mediaMap));

  rmSync(dir, { recursive: true, force: true });
  return zipSync(files);
}
```

```ts
import Database from "better-sqlite3";
import { unzipSync } from "fflate";
import { mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";

export class ApkgError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ApkgError";
  }
}

export interface ApkgNoteType {
  id: number;
  name: string;
  fields: string[];
  templates: { name: string; ord: number }[];
}

export interface ApkgNote {
  id: number;
  noteTypeId: number;
  fields: string[];
  tags: string[];
}

export interface ApkgCard {
  noteId: number;
  ord: number;
  type: number;
  due: number;
  ivl: number;
  factor: number;
  reps: number;
  lapses: number;
}

export interface ApkgContent {
  deckName: string;
  noteTypes: ApkgNoteType[];
  notes: ApkgNote[];
  cards: ApkgCard[];
  media: Map<string, Uint8Array>;
}

/** Anki joins a note's fields with the unit separator U+001F. */
const FIELD_SEPARATOR = String.fromCharCode(0x1f);

function tableExists(db: Database.Database, name: string): boolean {
  const row = db.prepare("SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = ?").get(name);
  return row !== undefined;
}

function readNoteTypes(db: Database.Database): ApkgNoteType[] {
  const col = db.prepare("SELECT models FROM col LIMIT 1").get() as { models: string } | undefined;

  if (col?.models && col.models !== "{}") {
    const parsed = JSON.parse(col.models) as Record<
      string,
      { id: number; name: string; flds: { name: string; ord: number }[]; tmpls: { name: string; ord: number }[] }
    >;
    return Object.values(parsed).map((model) => ({
      id: model.id,
      name: model.name,
      fields: [...model.flds].sort((a, b) => a.ord - b.ord).map((f) => f.name),
      templates: [...model.tmpls].sort((a, b) => a.ord - b.ord).map((t) => ({ name: t.name, ord: t.ord })),
    }));
  }

  if (!tableExists(db, "notetypes")) {
    throw new ApkgError("Die Sammlung enthält keine Notiztypen.");
  }

  const types = db.prepare("SELECT id, name FROM notetypes").all() as { id: number; name: string }[];
  return types.map((type) => ({
    id: type.id,
    name: type.name,
    fields: (
      db.prepare("SELECT name FROM fields WHERE ntid = ? ORDER BY ord").all(type.id) as { name: string }[]
    ).map((f) => f.name),
    templates: (
      db.prepare("SELECT name, ord FROM templates WHERE ntid = ? ORDER BY ord").all(type.id) as {
        name: string;
        ord: number;
      }[]
    ).map((t) => ({ name: t.name, ord: t.ord })),
  }));
}

function readDeckName(db: Database.Database): string {
  const col = db.prepare("SELECT decks FROM col LIMIT 1").get() as { decks: string } | undefined;
  if (col?.decks && col.decks !== "{}") {
    const parsed = JSON.parse(col.decks) as Record<string, { id: number; name: string }>;
    const named = Object.values(parsed).filter((deck) => deck.name !== "Default");
    if (named[0]) return named[0].name;
  }
  if (tableExists(db, "decks")) {
    const row = db.prepare("SELECT name FROM decks WHERE name != 'Default' ORDER BY id LIMIT 1").get() as
      | { name: string }
      | undefined;
    // Schema 18 stores nested deck names separated by U+001F rather than "::".
    if (row) return row.name.split(FIELD_SEPARATOR).join("::");
  }
  return "Importiertes Deck";
}

export function readApkg(bytes: Uint8Array): ApkgContent {
  let files: Record<string, Uint8Array>;
  try {
    files = unzipSync(bytes);
  } catch {
    throw new ApkgError("Die Datei ist kein gültiges Zip-Archiv und damit keine .apkg-Datei.");
  }

  if (files["collection.anki21b"] && !files["collection.anki21"] && !files["collection.anki2"]) {
    throw new ApkgError(
      "Diese Datei nutzt das neue komprimierte Anki-Format. Exportiere sie in Anki noch einmal mit der Option „Unterstützung für ältere Anki-Versionen\".",
    );
  }

  const collection = files["collection.anki21"] ?? files["collection.anki2"];
  if (!collection) {
    throw new ApkgError("Das Archiv enthält keine Anki-Sammlung (collection.anki2 oder collection.anki21).");
  }

  const dir = mkdtempSync(join(tmpdir(), "apkg-read-"));
  const dbPath = join(dir, "collection.db");
  writeFileSync(dbPath, collection);

  try {
    const db = new Database(dbPath, { readonly: true });
    try {
      const noteTypes = readNoteTypes(db);
      const deckName = readDeckName(db);

      const notes = (
        db.prepare("SELECT id, mid, flds, tags FROM notes ORDER BY id").all() as {
          id: number;
          mid: number;
          flds: string;
          tags: string;
        }[]
      ).map((row) => ({
        id: row.id,
        noteTypeId: row.mid,
        fields: row.flds.split(FIELD_SEPARATOR),
        tags: row.tags.trim().split(/\s+/).filter(Boolean),
      }));

      const cards = (
        db
          .prepare("SELECT nid, ord, type, due, ivl, factor, reps, lapses FROM cards ORDER BY nid, ord")
          .all() as {
          nid: number;
          ord: number;
          type: number;
          due: number;
          ivl: number;
          factor: number;
          reps: number;
          lapses: number;
        }[]
      ).map((row) => ({
        noteId: row.nid,
        ord: row.ord,
        type: row.type,
        due: row.due,
        ivl: row.ivl,
        factor: row.factor,
        reps: row.reps,
        lapses: row.lapses,
      }));

      const media = new Map<string, Uint8Array>();
      const mediaIndex = files["media"];
      if (mediaIndex) {
        const mapping = JSON.parse(new TextDecoder().decode(mediaIndex)) as Record<string, string>;
        for (const [number, filename] of Object.entries(mapping)) {
          const payload = files[number];
          if (payload) media.set(filename, payload);
        }
      }

      return { deckName, noteTypes, notes, cards, media };
    } finally {
      db.close();
    }
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }
}
```

```ts
import type { ApkgNoteType } from "./read.js";

export interface FieldMapping {
  front: number;
  back: number;
  reading: number | null;
  note: number | null;
}

const FRONT = ["expression", "ausdruck", "front", "vorderseite", "word", "wort", "kanji", "japanese", "japanisch"];
const BACK = ["meaning", "bedeutung", "back", "rückseite", "ruckseite", "translation", "übersetzung", "deutsch", "english"];
const READING = ["reading", "lesung", "kana", "furigana", "pronunciation", "aussprache"];
const NOTE = ["notes", "note", "notiz", "hinweis", "comment", "kommentar"];

function findField(fields: string[], needles: string[], exclude: number[]): number | null {
  for (let i = 0; i < fields.length; i += 1) {
    if (exclude.includes(i)) continue;
    const name = fields[i]!.toLowerCase();
    if (needles.some((needle) => name.includes(needle))) return i;
  }
  return null;
}

export function guessMapping(noteType: ApkgNoteType): FieldMapping {
  const fields = noteType.fields;
  if (fields.length === 0) return { front: 0, back: 0, reading: null, note: null };
  if (fields.length === 1) return { front: 0, back: 0, reading: null, note: null };

  const front = findField(fields, FRONT, []) ?? 0;
  const back = findField(fields, BACK, [front]) ?? (front === 0 ? 1 : 0);
  const reading = findField(fields, READING, [front, back]);
  const note = findField(fields, NOTE, [front, back, ...(reading === null ? [] : [reading])]);

  return { front, back, reading, note };
}
```

```ts
import { newSchedulingState, type SchedulingState } from "@skills/core";
import { createHash } from "node:crypto";
import { mkdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import type { AppDatabase } from "../../db/client.js";
import type { FieldMapping } from "./map.js";
import type { ApkgContent } from "./read.js";

export interface ImportOptions {
  skillId: string;
  moduleId: string;
  moduleTitle: string;
  mapping: Record<string, FieldMapping>;
  keepScheduling: boolean;
  mediaDir: string;
}

export interface ImportResult {
  notes: number;
  cards: number;
  media: number;
}

const DAY = 86_400_000;

/** Strips Anki's HTML and sound references so plain text remains. */
function clean(value: string): string {
  return value
    .replace(/\[sound:[^\]]+\]/g, "")
    .replace(/<br\s*\/?>/gi, " ")
    .replace(/<[^>]+>/g, "")
    .replace(/&nbsp;/g, " ")
    .trim();
}

function schedulingFromAnki(card: ApkgContent["cards"][number], now: number): SchedulingState {
  const stability = Math.max(0.1, card.ivl);
  const difficulty = Math.min(10, Math.max(1, 11 - card.factor / 400));
  return {
    state: "review",
    due: now + Math.max(0, card.ivl) * DAY,
    stability,
    difficulty,
    reps: card.reps,
    lapses: card.lapses,
    lastReview: now,
  };
}

export function importApkg(
  database: AppDatabase,
  content: ApkgContent,
  options: ImportOptions,
  now: number,
): ImportResult {
  const { raw } = database;
  const notesById = new Map(content.notes.map((note) => [note.id, note]));
  const result: ImportResult = { notes: 0, cards: 0, media: 0 };

  const run = raw.transaction(() => {
    raw
      .prepare("INSERT INTO skills (id, installed_at) VALUES (?, ?) ON CONFLICT(id) DO NOTHING")
      .run(options.skillId, now);
    raw
      .prepare(
        `INSERT INTO modules (id, skill_id, type, title) VALUES (?, ?, 'srs-deck', ?)
         ON CONFLICT(skill_id, id) DO UPDATE SET title = excluded.title`,
      )
      .run(options.moduleId, options.skillId, options.moduleTitle);

    const insertNote = raw.prepare(
      "INSERT INTO notes (skill_id, fields_json, source_apkg) VALUES (?, ?, ?)",
    );
    const insertCard = raw.prepare(
      `INSERT INTO cards (skill_id, module_id, card_key, note_id, front_json, back_json, tags, state)
       VALUES (?, ?, ?, ?, ?, ?, ?, 'active')
       ON CONFLICT(skill_id, module_id, card_key) DO UPDATE SET
         front_json = excluded.front_json, back_json = excluded.back_json,
         tags = excluded.tags, state = 'active'`,
    );
    const insertScheduling = raw.prepare(
      `INSERT INTO scheduling (card_id, state, due, stability, difficulty, reps, lapses, last_review)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?) ON CONFLICT(card_id) DO NOTHING`,
    );
    const findCard = raw.prepare(
      "SELECT id FROM cards WHERE skill_id = ? AND module_id = ? AND card_key = ?",
    );

    for (const note of content.notes) {
      insertNote.run(options.skillId, JSON.stringify(note.fields), options.moduleTitle);
      result.notes += 1;
    }

    for (const card of content.cards) {
      const note = notesById.get(card.noteId);
      if (!note) continue;
      const mapping = options.mapping[String(note.noteTypeId)];
      if (!mapping) continue;

      const front = clean(note.fields[mapping.front] ?? "");
      const back = clean(note.fields[mapping.back] ?? "");
      if (front === "" || back === "") continue;

      const reading = mapping.reading === null ? null : clean(note.fields[mapping.reading] ?? "") || null;
      const noteText = mapping.note === null ? null : clean(note.fields[mapping.note] ?? "") || null;
      const cardKey = `apkg-${note.id}-${card.ord}`;

      insertCard.run(
        options.skillId,
        options.moduleId,
        cardKey,
        null,
        JSON.stringify({ text: front }),
        JSON.stringify({ text: back, reading, note: noteText, example: null }),
        JSON.stringify(note.tags),
      );

      const row = findCard.get(options.skillId, options.moduleId, cardKey) as { id: number };
      const state =
        options.keepScheduling && card.type === 2 ? schedulingFromAnki(card, now) : newSchedulingState(now);
      insertScheduling.run(
        row.id,
        state.state,
        state.due,
        state.stability,
        state.difficulty,
        state.reps,
        state.lapses,
        state.lastReview,
      );
      result.cards += 1;
    }

    if (content.media.size > 0) {
      mkdirSync(options.mediaDir, { recursive: true });
      const insertMedia = raw.prepare(
        "INSERT INTO media (filename, mime, sha256, path) VALUES (?, ?, ?, ?) ON CONFLICT(sha256) DO NOTHING",
      );
      for (const [filename, bytes] of content.media) {
        const sha = createHash("sha256").update(bytes).digest("hex");
        const target = join(options.mediaDir, `${sha}-${filename}`);
        writeFileSync(target, bytes);
        insertMedia.run(filename, "application/octet-stream", sha, target);
        result.media += 1;
      }
    }
  });

  run();
  return result;
}
```

```ts
    const result = importApkg(database, content, { ...parsed.data, mediaDir }, Date.now());
    return c.json(result);
  } catch (error) {
    if (error instanceof ApkgError) {
      return c.json({ error: { code: "apkg_invalid", message: error.message } }, 400);
    }
    throw error;
  }
```

```tsx
      setModuleTitle(result.deckName);

        <p className="rounded border border-emerald-300 bg-emerald-50 p-3 text-sm text-emerald-800 dark:border-emerald-900 dark:bg-emerald-950 dark:text-emerald-200">
```

```ts
import Database from "better-sqlite3";
import { zipSync } from "fflate";
import { mkdtempSync, readFileSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import type { AppDatabase } from "../../db/client.js";
import { ApkgError } from "./read.js";

const SCHEMA = `
CREATE TABLE col (id INTEGER PRIMARY KEY, crt INTEGER, mod INTEGER, scm INTEGER,
  ver INTEGER, dty INTEGER, usn INTEGER, ls INTEGER, conf TEXT, models TEXT,
  decks TEXT, dconf TEXT, tags TEXT);
CREATE TABLE notes (id INTEGER PRIMARY KEY, guid TEXT, mid INTEGER, mod INTEGER,
  usn INTEGER, tags TEXT, flds TEXT, sfld TEXT, csum INTEGER, flags INTEGER, data TEXT);
CREATE TABLE cards (id INTEGER PRIMARY KEY, nid INTEGER, did INTEGER, ord INTEGER,
  mod INTEGER, usn INTEGER, type INTEGER, queue INTEGER, due INTEGER, ivl INTEGER,
  factor INTEGER, reps INTEGER, lapses INTEGER, left INTEGER, odue INTEGER,
  odid INTEGER, flags INTEGER, data TEXT);
CREATE TABLE revlog (id INTEGER PRIMARY KEY, cid INTEGER, usn INTEGER, ease INTEGER,
  ivl INTEGER, lastIvl INTEGER, factor INTEGER, time INTEGER, type INTEGER);
CREATE TABLE graves (usn INTEGER, oid INTEGER, type INTEGER);
`;

const NOTE_TYPE_ID = 1_500_000_000_000;
const DECK_ID = 1_500_000_000_001;
/** Anki joins a note's fields with the unit separator U+001F. */
const FIELD_SEPARATOR = String.fromCharCode(0x1f);

interface CardRow {
  card_key: string;
  front_json: string;
  back_json: string;
  tags: string;
}

export function writeApkg(
  database: AppDatabase,
  skillId: string,
  moduleId: string,
  deckName: string,
): Uint8Array {
  const rows = database.raw
    .prepare(
      "SELECT card_key, front_json, back_json, tags FROM cards WHERE skill_id = ? AND module_id = ? AND state = 'active' ORDER BY id",
    )
    .all(skillId, moduleId) as CardRow[];

  if (rows.length === 0) {
    throw new ApkgError("Dieses Modul enthält keine Karten zum Exportieren.");
  }

  const dir = mkdtempSync(join(tmpdir(), "apkg-write-"));
  const dbPath = join(dir, "collection.anki2");

  try {
    const db = new Database(dbPath);
    db.exec(SCHEMA);

    const models = JSON.stringify({
      [String(NOTE_TYPE_ID)]: {
        id: NOTE_TYPE_ID,
        name: "Skills-Export",
        type: 0,
        flds: ["Vorderseite", "Rückseite", "Lesung", "Notiz"].map((name, ord) => ({ name, ord })),
        tmpls: [
          {
            name: "Karte 1",
            ord: 0,
            qfmt: "{{Vorderseite}}",
            afmt: "{{FrontSide}}<hr id=answer>{{Rückseite}}<br>{{Lesung}}<br>{{Notiz}}",
          },
        ],
      },
    });

    const decks = JSON.stringify({ [String(DECK_ID)]: { id: DECK_ID, name: deckName } });
    const now = Date.now();

    db.prepare(
      "INSERT INTO col (id, crt, mod, scm, ver, dty, usn, ls, conf, models, decks, dconf, tags) VALUES (1, ?, ?, ?, 11, 0, 0, 0, '{}', ?, ?, '{}', '{}')",
    ).run(Math.floor(now / 1000), now, now, models, decks);

    const insertNote = db.prepare(
      "INSERT INTO notes (id, guid, mid, mod, usn, tags, flds, sfld, csum, flags, data) VALUES (?, ?, ?, ?, -1, ?, ?, ?, 0, 0, '')",
    );
    const insertCard = db.prepare(
      `INSERT INTO cards (id, nid, did, ord, mod, usn, type, queue, due, ivl, factor, reps, lapses, left, odue, odid, flags, data)
       VALUES (?, ?, ?, 0, ?, -1, 0, 0, ?, 0, 2500, 0, 0, 0, 0, 0, 0, '')`,
    );

    rows.forEach((row, i) => {
      const front = JSON.parse(row.front_json) as { text: string };
      const back = JSON.parse(row.back_json) as { text: string; reading: string | null; note: string | null };
      const tags = (JSON.parse(row.tags) as string[]).join(" ");
      const noteId = now + i;
      const fields = [front.text, back.text, back.reading ?? "", back.note ?? ""].join(FIELD_SEPARATOR);

      insertNote.run(noteId, row.card_key, NOTE_TYPE_ID, now, tags ? ` ${tags} ` : "", fields, front.text);
      insertCard.run(noteId + 1_000_000, noteId, DECK_ID, now, i);
    });

    db.close();

    return zipSync({
      "collection.anki2": new Uint8Array(readFileSync(dbPath)),
      media: new TextEncoder().encode("{}"),
    });
  } finally {
    rmSync(dir, { recursive: true, force: true });
  }
}
```

```ts
ankiRoutes.get("/api/anki/export/:skillId/:moduleId", (c) => {
  const database = c.get("database");
  const skillId = c.req.param("skillId");
  const moduleId = c.req.param("moduleId");

  const row = database.raw
    .prepare("SELECT title FROM modules WHERE skill_id = ? AND id = ?")

  const deckName = row?.title ?? moduleId;

  try {

      "content-disposition": `attachment; filename="${moduleId}.apkg"`,

    if (error instanceof ApkgError) {

  }
});
```
